🚀 Starting point

Starting point. A tensor is a block of memory plus a shape and a stride: element \((i,j)\) lives at offset \(i\cdot s_0+j\cdot s_1\). Transposing swaps the strides without moving data. Most PyTorch confusion (view vs reshape, contiguous, squeeze) is about this one fact.

Operation What it really does
x.T, transpose, permute new strides, same memory
view new shape, same memory; fails if the strides don’t allow it
reshape view if possible, otherwise copies
squeeze / unsqueeze remove / add a size-1 dimension
flatten(1) merge all dims from 1 onward, e.g. \((B,C,H,W)\to(B,CHW)\)
cat / stack join along an existing / a new dimension
gather(dim, idx) pick one entry per row, e.g. the target token’s log-prob
a @ b batched matmul; leading dims broadcast

The seven tasks at the end of the script are the ones you actually write when building a transformer: a causal mask, one-hot targets, batched outer products, top-\(k\) logit filtering for sampling, a padding mask from lengths, attention with einsum (checked against PyTorch’s fused kernel), and repeat_interleave to expand grouped-query KV heads.

"""Toolkit ch. 3: PyTorch tensor operations, plus seven transformer-flavoured tasks."""
import torch

torch.manual_seed(0)

# Creation and special tensors.
print("[create]", torch.arange(0, 10, 3).tolist(), torch.linspace(0, 1, 5).tolist())
print("[create] eye(3) diag:", torch.eye(3).diagonal().tolist(), " zeros_like:", torch.zeros_like(torch.eye(2)).shape)

# Strides: view needs contiguous memory, reshape copies when it must.
x = torch.arange(12).reshape(3, 4)
xt = x.T
print(f"[stride] x {tuple(x.shape)} stride {x.stride()} | x.T stride {xt.stride()} contiguous={xt.is_contiguous()}")
try:
    xt.view(12)
except RuntimeError:
    print("[stride] x.T.view(12) -> RuntimeError; x.T.reshape(12) ->", xt.reshape(12)[:6].tolist())

# squeeze / unsqueeze / flatten.
v = torch.randn(4)
print(f"[shape] unsqueeze(0) {tuple(v.unsqueeze(0).shape)}  unsqueeze(1) {tuple(v.unsqueeze(1).shape)}"
      f"  squeeze {tuple(v[None, :, None].squeeze().shape)}  flatten {tuple(torch.randn(2, 3, 4).flatten(1).shape)}")

# cat vs stack.
a, b = torch.randn(2, 3), torch.randn(2, 3)
print(f"[join] cat dim0 {tuple(torch.cat([a, b]).shape)}  stack dim0 {tuple(torch.stack([a, b]).shape)}")

# Indexing: gather picks the logit of each target token -> cross-entropy by hand.
logits, y = torch.randn(4, 10), torch.tensor([3, 0, 9, 1])
nll = -(logits.log_softmax(-1).gather(1, y[:, None]).squeeze(1)).mean()
print(f"[gather] manual CE {nll:.5f}  F.cross_entropy {torch.nn.functional.cross_entropy(logits, y):.5f}")

# Batched matmul broadcasts the batch dims: (2,1,3,4) @ (5,4,6) -> (2,5,3,6).
print("[matmul]", tuple((torch.randn(2, 1, 3, 4) @ torch.randn(5, 4, 6)).shape))

# Seven tasks.
T = 5
causal = torch.tril(torch.ones(T, T, dtype=torch.bool))                        # 1 causal mask
onehot = torch.nn.functional.one_hot(torch.tensor([2, 0, 1]), 4)              # 2 one-hot
outer = torch.einsum("bi,bj->bij", torch.randn(3, 4), torch.randn(3, 2))       # 3 batched outer product
z = torch.randn(2, 8)
kth = z.topk(3, dim=-1).values[:, -1:]
topk = z.masked_fill(z < kth, float("-inf"))                                 # 4 top-k logits mask
lens = torch.tensor([3, 5])
pad = torch.arange(T)[None, :] < lens[:, None]                               # 5 padding mask
Q, K, V = (torch.randn(2, 4, T, 8) for _ in range(3))
att = torch.einsum("bhqd,bhkd->bhqk", Q, K) / 8**0.5
att = att.masked_fill(~causal, float("-inf")).softmax(-1)
out = torch.einsum("bhqk,bhkd->bhqd", att, V)                                 # 6 attention with einsum
ref = torch.nn.functional.scaled_dot_product_attention(Q, K, V, is_causal=True)
kv = torch.randn(2, 2, T, 8).repeat_interleave(2, dim=1)                      # 7 GQA: 2 kv heads -> 4
print(f"[tasks] causal row2 {causal[2].int().tolist()}  onehot {onehot[0].tolist()}  outer {tuple(outer.shape)}")
print(f"[tasks] top-k kept per row {(topk > float('-inf')).sum(1).tolist()}  pad {pad.int().tolist()}")
print(f"[tasks] einsum attention == SDPA: {torch.allclose(out, ref, atol=1e-5)}  GQA kv {tuple(kv.shape)}")

The same ideas in Keras

The TensorFlow track of the curriculum has 27 lessons, but most of them are the same models in a different API: linear models, CNNs, save/restore, transfer learning, fine-tuning, adversarial noise, DeepDream and style transfer, input pipelines, time series. Three translation rules cover almost everything:

PyTorch TensorFlow / Keras
nn.Module.forward keras.Model / functional API
loss.backward() tf.GradientTape().gradient(loss, vars)
DataLoader tf.data.Dataset.shuffle().batch().prefetch()
torch.save(state_dict) model.save("m.keras")
requires_grad_(False) layer.trainable = False
torch.compile @tf.function

Adversarial examples, DeepDream and style transfer share one trick: take the gradient with respect to the input instead of the weights. FGSM moves the image by \(\epsilon\operatorname{sign}(\nabla_xL)\), the largest loss increase allowed inside an \(L_\infty\) ball of radius \(\epsilon\) (to first order, \(\max_{\lVert\delta\rVert_\infty\le\epsilon}\nabla_xL\cdot\delta=\epsilon\lVert\nabla_xL\rVert_1\), reached at the sign vector). DeepDream ascends a layer’s activation norm; style transfer descends a loss on Gram matrices of features.

"""Toolkit ch. 3b: the same ideas in TensorFlow/Keras. NOT run here (TensorFlow is not installed);
written against the Keras 3 API (pip install tensorflow==2.20.0)."""
import tensorflow as tf
from tensorflow import keras

# 1. tf.data: a streaming input pipeline (the TFRecords / Dataset API lessons).
(x_train, y_train), (x_test, y_test) = keras.datasets.mnist.load_data()
x_train, x_test = x_train[..., None] / 255.0, x_test[..., None] / 255.0
train = (tf.data.Dataset.from_tensor_slices((x_train, y_train))
         .shuffle(10_000).batch(128).prefetch(tf.data.AUTOTUNE))

# 2. A small CNN with the Keras functional API.
inp = keras.Input((28, 28, 1))
h = keras.layers.Conv2D(32, 3, activation="relu")(inp)
h = keras.layers.MaxPooling2D()(h)
h = keras.layers.Conv2D(64, 3, activation="relu")(h)
h = keras.layers.GlobalAveragePooling2D()(h)
out = keras.layers.Dense(10)(h)                      # logits
model = keras.Model(inp, out)
model.compile(optimizer=keras.optimizers.AdamW(1e-3),
              loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
              metrics=["accuracy"])
model.fit(train, epochs=2, validation_data=(x_test, y_test))

# 3. Save and restore.
model.save("mnist_cnn.keras")
model = keras.models.load_model("mnist_cnn.keras")


# 4. A custom training step with GradientTape (what model.fit does inside).
@tf.function
def train_step(x, y):
    with tf.GradientTape() as tape:
        loss = model.compiled_loss(y, model(x, training=True))
    grads = tape.gradient(loss, model.trainable_variables)
    model.optimizer.apply_gradients(zip(grads, model.trainable_variables))
    return loss


# 5. Adversarial example (FGSM): step the *input* along the sign of the loss gradient.
def fgsm(x, y, eps=0.1):
    x = tf.convert_to_tensor(x, tf.float32)
    with tf.GradientTape() as tape:
        tape.watch(x)
        loss = keras.losses.sparse_categorical_crossentropy(y, model(x), from_logits=True)
    return tf.clip_by_value(x + eps * tf.sign(tape.gradient(loss, x)), 0.0, 1.0)


# 6. Transfer learning: freeze a pretrained backbone, train a new head, then unfreeze to fine-tune.
base = keras.applications.MobileNetV3Small(include_top=False, weights="imagenet", pooling="avg")
base.trainable = False
clf = keras.Sequential([base, keras.layers.Dense(5)])
