🚀 Starting point

RLVR: sample several answers per problem, check each with a program (exact match, unit tests), and reinforce the ones that pass. GRPO compares answers to the same prompt instead of training a critic. TRL’s GRPOTrainer implements it, with options for every fix from the RL book’s Part XVIII: loss_type="dapo" (the default), "dr_grpo", "cispo"; epsilon_high (clip-higher); importance_sampling_level="sequence" (GSPO); scale_rewards="batch" (REINFORCE++-style).

One GRPO iteration. With vllm_mode="colocate", generation and training share the same GPU and weights are synced in memory.

✏️ Derivation: what makes a good RLVR dataset

Step 1. A group gives signal only if its rewards differ (RL book, Topic 32): \(P(\text{signal})=1-p^G-(1-p)^G\) for solve rate \(p\).

Step 2. So filter prompts to the model’s frontier: \(0.1\lesssim p\lesssim0.9\). Estimate \(p\) with 8 samples per prompt before training and drop the rest.

Step 3. The verifier is the objective. boxed() takes the last \boxed{} and compares numbers, not strings, so “72”, “72.0” and “7,2”-style formatting are handled consistently (see the verifier output below).

Step 4. The small format reward keeps early training from stalling when no answer is boxed yet. Keep it small: it’s a shaping term (RL book, Topic 62).

"""RL with verifiable rewards: GRPO on GSM8K with TRL and vLLM.

Local smoke test:  python grpo.py                   (tiny model, CPU, reward functions checked first)
On Modal:          modal run grpo.py::train_remote  (Qwen3-1.7B, 1x H100, vLLM colocated)
"""
import re

import modal

SYSTEM = "Solve the problem. Think step by step, then give the final answer as \\boxed{number}."


def text_of(c):
    return c if isinstance(c, str) else c[0]["content"]          # standard vs conversational format


def boxed(text: str) -> str | None:
    m = re.findall(r"\\boxed\{([^{}]*)\}", text)
    return m[-1].replace(",", "").strip() if m else None


def correctness_reward(completions, answer, **kwargs) -> list[float]:
    """1 if the last \\boxed{} matches the reference number, else 0. This is the verifier."""
    out = []
    for c, ref in zip(completions, answer):
        pred = boxed(text_of(c))
        try:
            out.append(1.0 if pred is not None and abs(float(pred) - float(ref)) < 1e-6 else 0.0)
        except ValueError:
            out.append(0.0)
    return out


def format_reward(completions, **kwargs) -> list[float]:
    """A small bonus for producing exactly one \\boxed{} answer."""
    return [0.1 if text_of(c).count("\\boxed{") == 1 else 0.0 for c in completions]


def gsm8k_prompts(split="train"):
    from datasets import load_dataset

    ds = load_dataset("openai/gsm8k", "main", split=split)
    return ds.map(lambda x: {
        "prompt": [{"role": "system", "content": SYSTEM}, {"role": "user", "content": x["question"]}],
        "answer": x["answer"].split("####")[-1].replace(",", "").strip(),   # reference after '####'
    }, remove_columns=ds.column_names)


def run_grpo(model, dataset, output_dir, **overrides):
    from trl import GRPOConfig, GRPOTrainer

    defaults = dict(
        output_dir=output_dir,
        num_generations=8,                   # G responses per prompt -> group-relative advantages
        max_completion_length=512,
        per_device_train_batch_size=16,      # must be divisible by num_generations
        gradient_accumulation_steps=4,
        learning_rate=1e-6,
        beta=0.0,                            # no KL term, no reference model (TRL default)
        loss_type="dapo",                    # token-level normalization (TRL default)
        epsilon=0.2, epsilon_high=0.28,      # DAPO clip-higher
        logging_steps=5,
        save_strategy="no",
        report_to="none",
    )
    args = GRPOConfig(**{**defaults, **overrides})   # smoke tests override some defaults
    trainer = GRPOTrainer(model=model, reward_funcs=[correctness_reward, format_reward],
                          args=args, train_dataset=dataset)
    trainer.train()
    return trainer


app = modal.App("pt-book-grpo")
image = modal.Image.debian_slim(python_version="3.12").uv_pip_install(
    "trl[vllm]==1.16.0", "transformers==5.20.2", "datasets==5.1.0", "accelerate==1.15.0"
)
models = modal.Volume.from_name("pt-book-models", create_if_missing=True)


@app.function(gpu="H100", image=image, volumes={"/models": models},
              secrets=[modal.Secret.from_name("huggingface-secret")], timeout=8 * 60 * 60)
def train_remote(max_steps: int = 500):
    trainer = run_grpo("Qwen/Qwen3-1.7B", gsm8k_prompts(), "/models/qwen3-1.7b-grpo", bf16=True,
                       max_steps=max_steps, use_vllm=True, vllm_mode="colocate")
    trainer.save_model("/models/qwen3-1.7b-grpo")
    models.commit()
