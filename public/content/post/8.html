🚀 Starting point

After SFT, the model follows instructions but its answers vary in quality. Preference data pairs a prompt with a chosen and a rejected response. DPO raises the chosen response’s log-probability relative to a frozen reference model and lowers the rejected one’s, with no reward model and no sampling (derivation: LLM Fundamentals, Session 26).

✏️ Derivation: reading DPO’s training logs

Step 1. TRL logs the implicit rewards \(\hat r^\pm=\beta\big(\log\pi_\theta(y^\pm|x)-\log\pi_{\text{ref}}(y^\pm|x)\big)\).

Step 2. rewards/margins \(=\overline{\hat r^+-\hat r^-}\). It starts at 0 (policy = reference) and should grow.

Step 3. rewards/accuracies = fraction of pairs with \(\hat r^+>\hat r^-\). Starts near 0.5.

Step 4. Loss \(=\overline{\softplus(-(\hat r^+-\hat r^-))}\), starting at \(\ln2=0.693\).

Step 5. Watch out: \(\hat r^+\) often goes down too (both responses become less likely, the rejected one more so). Mild is normal; large drops in \(\hat r^+\) mean the model is drifting toward degenerate text, so lower the learning rate or raise \(\beta\).

"""Direct Preference Optimization with TRL on UltraFeedback.

Local smoke test:  python dpo.py                    (tiny model, CPU)
On Modal:          modal run dpo.py::train_remote   (starts from the SFT checkpoint, 1x H100)
"""
import modal


def run_dpo(model, dataset, output_dir, **overrides):
    from peft import LoraConfig
    from trl import DPOConfig, DPOTrainer

    defaults = dict(
        output_dir=output_dir,
        beta=0.1,                            # KL strength; smaller = further from the reference
        learning_rate=5e-6,
        per_device_train_batch_size=4,
        gradient_accumulation_steps=8,
        num_train_epochs=1,
        logging_steps=10,
        save_strategy="no",
        report_to="none",
    )
    args = DPOConfig(**{**defaults, **overrides})   # smoke tests override some defaults
    # With LoRA, the reference model is the base weights with the adapter disabled: no second copy in memory.
    peft = LoraConfig(r=16, lora_alpha=32, target_modules="all-linear", task_type="CAUSAL_LM")
    trainer = DPOTrainer(model=model, args=args, train_dataset=dataset, peft_config=peft)
    trainer.train()
    return trainer


app = modal.App("pt-book-dpo")
image = modal.Image.debian_slim(python_version="3.12").uv_pip_install(
    "torch==2.14.1", "transformers==5.20.2", "trl==1.16.0", "datasets==5.1.0", "peft==0.21.2", "accelerate==1.15.0"
)
models = modal.Volume.from_name("pt-book-models", create_if_missing=True)


@app.function(gpu="H100", image=image, volumes={"/models": models}, timeout=4 * 60 * 60)
def train_remote():
    from datasets import load_dataset

    ds = load_dataset("trl-lib/ultrafeedback_binarized", split="train")   # ~62K conversational pairs
    trainer = run_dpo("/models/qwen3-0.6b-sft", ds, "/models/qwen3-0.6b-dpo", bf16=True)
    trainer.save_model("/models/qwen3-0.6b-dpo")
    models.commit()


if __name__ == "__main__":
    from datasets import Dataset

    pairs = [{"prompt": f"Q: What is {a} times 2?\nA:", "chosen": f" {2 * a}.", "rejected": f" {2 * a + 1}."}
             for a in range(32)]
    trainer = run_dpo("trl-internal-testing/tiny-Qwen2ForCausalLM-2.5", Dataset.from_list(pairs), "/tmp/dpo_smoke",
                      per_device_train_batch_size=4, gradient_accumulation_steps=1, max_steps=6, learning_rate=1e-3,
                      logging_steps=2, bf16=False, use_cpu=True)
    for h in trainer.state.log_history:
        if "loss" in h:
            print({k: round(v, 3) for k, v in h.items() if k in ("loss", "rewards/margins", "rewards/accuracies")})
