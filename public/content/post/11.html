🚀 Starting point

A reward model (RM) is a language model with its output layer replaced by a single number. It’s trained on preference pairs with the Bradley–Terry loss \(-\log\sigma(r^+-r^-)\) (LLM Fundamentals, Session 36) and then used three ways: as the reward in PPO-style RLHF, to pick the best of \(N\) samples, and to label new preference pairs for DPO.

✏️ Derivation: why TRL adds a centering term

Step 1. Bradley–Terry only sees differences: replacing \(r(x,y)\) with \(r(x,y)+c(x)\) leaves the loss unchanged, so the scores’ offset drifts freely during training.

Step 2. A drifting offset is harmless for ranking but harmful for RL, where the absolute scale of the reward sets the effective KL strength (RL book, Topic 127).

Step 3. center_rewards_coefficient\(=\lambda\) adds \(\lambda\,\overline{(r^++r^-)^2}\), which pulls the mean score toward 0 without changing which response wins. TRL recommends \(\lambda=0.01\).

Step 4. Ensembles (several RMs, use the mean minus a multiple of their spread) reduce reward hacking but don’t eliminate it: members trained on the same data share blind spots (RL book, Topics 130–131).

✏️ Derivation: two reward models instead of one

Step 1. PKU-SafeRLHF labels each response pair twice: which is more helpful and which is safer. These often disagree: the safer answer is frequently a refusal.

Step 2. Training one RM on a mix of both labels forces it to pick one exchange rate between helpfulness and harmlessness, baked into the weights.

Step 3. Training two (\(R_{\text{help}}\) and \(C_{\text{harm}}\)) keeps that exchange rate explicit, so it can be set by a constraint during RL (Safe RLHF, Chapter 21).

"""Reward models and unpaired preferences: TRL's RewardTrainer and KTOTrainer, plus turning
PKU-SafeRLHF into separate helpfulness and harmlessness preference sets.

Local smoke tests:  python reward_kto.py
On Modal:           modal run reward_kto.py::train_rm_remote
"""
import modal

TINY = "trl-internal-testing/tiny-Qwen2ForCausalLM-2.5"


def run_reward_model(model, dataset, output_dir, **overrides):
    from trl import RewardConfig, RewardTrainer

    defaults = dict(
        output_dir=output_dir,
        max_length=1024,
        center_rewards_coefficient=0.01,     # keeps scores near 0: Bradley-Terry can't pin the offset
        learning_rate=1e-5,
        per_device_train_batch_size=8,
        num_train_epochs=1,
        logging_steps=10,
        save_strategy="no",
        report_to="none",
    )
    trainer = RewardTrainer(model=model, args=RewardConfig(**{**defaults, **overrides}), train_dataset=dataset)
    trainer.train()
    return trainer


def run_kto(model, dataset, output_dir, **overrides):
    from trl import KTOConfig, KTOTrainer

    defaults = dict(
        output_dir=output_dir,
        beta=0.1,
        desirable_weight=1.0,                # raise one weight if your thumbs-up/down counts are unbalanced
        undesirable_weight=1.0,
        learning_rate=5e-7,
        per_device_train_batch_size=8,
        num_train_epochs=1,
        logging_steps=10,
        save_strategy="no",
        report_to="none",
    )
    trainer = KTOTrainer(model=model, args=KTOConfig(**{**defaults, **overrides}), train_dataset=dataset)
    trainer.train()
    return trainer


def saferlhf_to_pairs(row, axis):
    """PKU-SafeRLHF labels each response pair twice: which is more helpful, which is safer.
    Split them into two preference sets so helpfulness and harmlessness get separate reward models."""
    best = row["better_response_id"] if axis == "helpful" else row["safer_response_id"]
    chosen, rejected = row[f"response_{best}"], row[f"response_{1 - best}"]
    return {"prompt": [{"role": "user", "content": row["prompt"]}],
            "chosen": [{"role": "assistant", "content": chosen}],
            "rejected": [{"role": "assistant", "content": rejected}]}


app = modal.App("pt-book-reward")
image = modal.Image.debian_slim(python_version="3.12").uv_pip_install(
    "torch==2.14.1", "transformers==5.20.2", "trl==1.16.0", "datasets==5.1.0", "accelerate==1.15.0"
)
models = modal.Volume.from_name("pt-book-models", create_if_missing=True)


@app.function(gpu="H100", image=image, volumes={"/models": models}, timeout=4 * 60 * 60)
def train_rm_remote(axis: str = "harmless"):
    from datasets import load_dataset

    ds = load_dataset("PKU-Alignment/PKU-SafeRLHF", split="train")
    ds = ds.map(lambda r: saferlhf_to_pairs(r, "safe" if axis == "harmless" else "helpful"),
                remove_columns=ds.column_names)
    trainer = run_reward_model("Qwen/Qwen3-0.6B", ds, f"/models/rm-{axis}", bf16=True)
    trainer.save_model(f"/models/rm-{axis}")
    models.commit()


if __name__ == "__main__":
    from datasets import Dataset

    row = {"prompt": "How do I pick a strong password?", "response_0": "Use 'password123'.",
           "response_1": "Use a long passphrase from a password manager.", "better_response_id": 1, "safer_response_id": 1}
    print("SafeRLHF -> harmless pair, chosen:", saferlhf_to_pairs(row, "safe")["chosen"][0]["content"])

    pairs = [{"prompt": f"What is {a} + {a}?", "chosen": f" {2 * a}", "rejected": f" {2 * a + 3}"} for a in range(32)]
    rm = run_reward_model(TINY, Dataset.from_list(pairs), "/tmp/rm_smoke", max_length=64, per_device_train_batch_size=4,
                          max_steps=8, logging_steps=4, learning_rate=1e-3, bf16=False, use_cpu=True)
    for h in rm.state.log_history:
        if "loss" in h:
            print("RM  ", {k: round(h[k], 3) for k in ("loss", "accuracy", "margin") if k in h})

    unpaired = [{"prompt": f"What is {a} + {a}?", "completion": f" {2 * a + (a % 2) * 3}", "label": a % 2 == 0}
                for a in range(32)]                                   # True = thumbs up, False = thumbs down
    kto = run_kto(TINY, Dataset.from_list(unpaired), "/tmp/kto_smoke", per_device_train_batch_size=8,
                  max_steps=4, logging_steps=2, learning_rate=1e-4, bf16=False, use_cpu=True)
    for h in kto.state.log_history:
        if "loss" in h:
            print("KTO ", {k: round(h[k], 3) for k in ("loss", "kl") if k in h})
