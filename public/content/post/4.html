🚀 Starting point

The whole of pre-training is one loop: sample a batch of token sequences, compute the next-token loss (LLM Fundamentals, Session 11), backpropagate, clip gradients, and step AdamW with a learning-rate schedule. The model below is the LLaMA recipe at small scale: RMSNorm, SwiGLU, RoPE, tied embeddings, and scaled_dot_product_attention (FlashAttention on GPU).

"""Pre-train a small LLaMA-style GPT from scratch on FineWeb-Edu.

Local smoke test:   python pretrain_gpt.py                  (tiny model, CPU, 40 steps)
On Modal:           modal run pretrain_gpt.py::train_remote (124M-class model, 1x H100)
"""
import math
import time
from dataclasses import dataclass

import modal
import torch
import torch.nn as nn
import torch.nn.functional as F


@dataclass
class Config:
    vocab_size: int = 49152          # SmolLM2 tokenizer
    d_model: int = 768
    n_layers: int = 12
    n_heads: int = 12
    seq_len: int = 1024
    batch_size: int = 32             # sequences per step -> 32 x 1024 = 32,768 tokens/step
    max_steps: int = 10_000          # about 330M tokens
    lr: float = 1e-3
    warmup: int = 200
    decay_frac: float = 0.2          # WSD: constant LR, then linear decay over the last 20%
    weight_decay: float = 0.1


def rope_tables(seq_len, head_dim, base=10000.0, device=None):
    inv = base ** (-torch.arange(0, head_dim, 2, device=device).float() / head_dim)
    ang = torch.outer(torch.arange(seq_len, device=device).float(), inv)
    return ang.cos(), ang.sin()


def apply_rope(x, cos, sin):                         # x: [B, H, T, D]; rotate feature pairs
    x1, x2 = x[..., 0::2], x[..., 1::2]
    return torch.stack([x1 * cos - x2 * sin, x1 * sin + x2 * cos], dim=-1).flatten(-2)


class Block(nn.Module):
    def __init__(self, c: Config):
        super().__init__()
        d, self.h = c.d_model, c.n_heads
        hidden = 64 * round(8 * d / 3 / 64)          # SwiGLU width: 2/3 of 4d, rounded
        self.n1, self.n2 = nn.RMSNorm(d), nn.RMSNorm(d)
        self.qkv = nn.Linear(d, 3 * d, bias=False)
        self.o = nn.Linear(d, d, bias=False)
        self.w13 = nn.Linear(d, 2 * hidden, bias=False)
        self.w2 = nn.Linear(hidden, d, bias=False)

    def forward(self, x, cos, sin):
        B, T, D = x.shape
        q, k, v = self.qkv(self.n1(x)).view(B, T, 3, self.h, D // self.h).permute(2, 0, 3, 1, 4)
        q, k = apply_rope(q, cos, sin), apply_rope(k, cos, sin)
        y = F.scaled_dot_product_attention(q, k, v, is_causal=True)       # FlashAttention on GPU
        x = x + self.o(y.transpose(1, 2).reshape(B, T, D))
        a, b = self.w13(self.n2(x)).chunk(2, dim=-1)
        return x + self.w2(F.silu(a) * b)


class GPT(nn.Module):
    def __init__(self, c: Config):
        super().__init__()
        self.c = c
        self.emb = nn.Embedding(c.vocab_size, c.d_model)
        self.blocks = nn.ModuleList(Block(c) for _ in range(c.n_layers))
        self.norm = nn.RMSNorm(c.d_model)
        self.head = nn.Linear(c.d_model, c.vocab_size, bias=False)
        self.head.weight = self.emb.weight                                # weight tying
        for name, p in self.named_parameters():
            if p.dim() == 2:
                std = 0.02 / math.sqrt(2 * c.n_layers) if name.endswith(("o.weight", "w2.weight")) else 0.02
                nn.init.normal_(p, std=std)
        cos, sin = rope_tables(c.seq_len, c.d_model // c.n_heads)
        self.register_buffer("cos", cos, persistent=False)
        self.register_buffer("sin", sin, persistent=False)

    def forward(self, idx, targets=None):
        T = idx.shape[1]
        x = self.emb(idx)
        for blk in self.blocks:
            x = blk(x, self.cos[:T], self.sin[:T])
        logits = self.head(self.norm(x))
        if targets is None:
            return logits
        return F.cross_entropy(logits.float().reshape(-1, logits.size(-1)), targets.reshape(-1))


def packed_batches(texts, tokenizer, B, T):
    """Concatenate documents (EOS between them) and cut into [B, T+1] blocks: no padding, no waste."""
    buf = []
    for text in texts:
        buf += tokenizer(text)["input_ids"] + [tokenizer.eos_token_id]
        while len(buf) >= B * (T + 1):
            x = torch.tensor(buf[:B * (T + 1)]).view(B, T + 1)
            buf = buf[B * (T + 1):]
            yield x[:, :-1], x[:, 1:]


def lr_at(step, c: Config):
    if step < c.warmup:
        return c.lr * (step + 1) / c.warmup
    decay_start = int(c.max_steps * (1 - c.decay_frac))
    if step < decay_start:
        return c.lr
    return c.lr * max(0.0, 1 - (step - decay_start) / max(1, c.max_steps - decay_start))


def make_optimizer(model, c: Config, fused: bool):
    decay = [p for p in model.parameters() if p.dim() >= 2]       # matrices and embeddings
    no_decay = [p for p in model.parameters() if p.dim() < 2]     # norms
    return torch.optim.AdamW([{"params": decay, "weight_decay": c.weight_decay},
                              {"params": no_decay, "weight_decay": 0.0}],
                             lr=c.lr, betas=(0.9, 0.95), fused=fused)


def train(c: Config, texts, tokenizer, device="cpu", log_every=10, on_step=None):
    torch.manual_seed(0)
    model = GPT(c).to(device)
    n_params = sum(p.numel() for p in model.parameters())
    opt = make_optimizer(model, c, fused=device == "cuda")
    losses, t0 = [], time.time()
    for step, (x, y) in enumerate(packed_batches(texts, tokenizer, c.batch_size, c.seq_len)):
        if step >= c.max_steps:
            break
        for g in opt.param_groups:
            g["lr"] = lr_at(step, c)
        x, y = x.to(device), y.to(device)
        with torch.autocast(device_type=device, dtype=torch.bfloat16, enabled=device == "cuda"):
            loss = model(x, y)
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        opt.step()
        opt.zero_grad(set_to_none=True)
        losses.append(loss.item())
        if step % log_every == 0:
            dt = time.time() - t0
            tok_s = (step + 1) * c.batch_size * c.seq_len / dt
            mfu = 6 * n_params * tok_s / 989e12 if device == "cuda" else float("nan")   # H100 dense BF16 peak
            print(f"step {step:5d}  loss {loss.item():.3f}  lr {lr_at(step, c):.2e}  tok/s {tok_s:,.0f}  MFU {mfu:.2f}")
        if on_step:
            on_step(step, model)
    return model, losses

def train_remote(max_steps: int = 10_000):
    from datasets import load_dataset
    from transformers import AutoTokenizer

    torch.backends.cuda.matmul.allow_tf32 = True
    tok = AutoTokenizer.from_pretrained("HuggingFaceTB/SmolLM2-135M")
    stream = load_dataset("HuggingFaceFW/fineweb-edu", name="sample-10BT", split="train", streaming=True)
    c = Config(max_steps=max_steps)

    def save(step, model):
        if step > 0 and step % 2000 == 0:
            torch.save(model.state_dict(), f"/ckpt/gpt_step{step}.pt")
            ckpt.commit()

    train(c, (d["text"] for d in stream), tok, device="cuda", log_every=100, on_step=save)

✏️ Derivation: the pieces of the loop

Step 1. Packing: concatenate documents with an EOS between them and cut every \(T+1\) tokens. Inputs \(x_{0:T}\), targets \(x_{1:T+1}\). No padding, so every token trains. (Attention can cross the EOS into the previous document; at this scale that’s harmless, and larger runs mask it with document-aware attention.)

Step 2. Init: weights \(\mathcal N(0,0.02^2)\), and the two projections that write into the residual stream (o, w2) scaled by \(1/\sqrt{2L}\), so the residual stream’s variance after \(L\) blocks stays about the same as at the start (each block adds two terms).

Step 3. AdamW with \(\beta_2=0.95\) (not 0.999): a shorter memory for \(v\), which reacts faster to gradient spikes. Weight decay only on matrices and embeddings, never on norm gains.

Step 4. WSD schedule (warmup–stable–decay): linear warmup, a long constant phase, then a linear decay to 0 over the last 10–20%. Unlike cosine, you can stop the constant phase whenever you like and decay from any checkpoint, and the decay phase is where you put your best data (mid-training).

Step 5. MFU printed by the loop: \(6N\cdot\text{tokens/s}/989\text{T}\). On one H100 with BF16 and torch.compile off, a model this size typically lands around 30–45%.

The two common schedules. Most of WSD’s loss improvement arrives during the final decay.
